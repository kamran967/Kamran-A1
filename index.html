function localCommand(text){ const clean = text .trim() .replace(/[۔؟?.,،]/g, "") .replace(/\s+/g, " "); const lower = clean.toLowerCase(); /* ========================= NAME QUESTION ========================= */ if( clean.includes("میرا نام") && ( clean.includes("کیا ہے") || clean.includes("کون ہے") || clean.includes("بتاؤ") || clean.includes("بتاو") ) ){ const name = mem.find( x => x.type === "name" ); if(name && name.value){ return "آپ کا نام " + name.value + " ہے۔"; } return "آپ کا نام ابھی Memory میں محفوظ نہیں ہے۔"; } /* ========================= NAME SAVE ========================= */ if( clean.includes("میرا نام") ){ let name = ""; /* "میرا نام کامران ہے" */ let match = clean.match( /میرا نام\s+(.+?)(?:\s+ہے)?$/i ); if(match){ name = match[1].trim(); } /* اضافی الفاظ صاف کریں */ name = name .replace( /^(ہے|ہوں|سیو کرلو|save کرلو|یاد رکھو|یاد کرلو)\s*/i, "" ) .trim(); /* اگر نام مل گیا */ if( name && name !== "کیا ہے" && name !== "کون ہے" ){ mem = mem.filter( x => x.type !== "name" ); mem.push({ type:"name", value:name }); saveMemory(); return ( "ٹھیک ہے، میں آپ کا نام یاد رکھوں گا: " + name ); } } /* ========================= GENERAL MEMORY ========================= */ if( clean.startsWith("یاد رکھو") ){ const value = clean .replace( /^یاد رکھو[:：]?\s*/, "" ) .trim(); if(value){ mem.push({ type:"note", value:value }); saveMemory(); return "ٹھیک ہے، یہ بات Memory میں محفوظ کر لی ہے۔"; } } return null; }
